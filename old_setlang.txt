    // Form success message
    const succ = document.getElementById('form-success');
    if (succ) {
      const esSpan = succ.querySelector('.es');
      const enSpan = succ.querySelector('.en');
      if (esSpan) esSpan.style.display = lang === 'es' ? 'block' : 'none';
      if (enSpan) enSpan.style.display = lang === 'en' ? 'block' : 'none';
    }
